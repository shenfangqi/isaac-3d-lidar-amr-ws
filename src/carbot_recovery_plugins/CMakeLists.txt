cmake_minimum_required(VERSION 3.8)
project(carbot_recovery_plugins)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(deps rclcpp rclcpp_action rclcpp_lifecycle nav2_core nav2_util
  nav2_costmap_2d nav2_behavior_tree behaviortree_cpp_v3
  nav2_regulated_pure_pursuit_controller pluginlib tf2_geometry_msgs
  geometry_msgs nav_msgs carbot_recovery_interfaces)
foreach(dep ${deps})
  find_package(${dep} REQUIRED)
endforeach()
add_library(carbot_recovery_plugins SHARED src/controller.cpp src/evidence_layer.cpp src/behavior.cpp)
target_include_directories(carbot_recovery_plugins PRIVATE include)
ament_target_dependencies(carbot_recovery_plugins ${deps})
add_library(carbot_bounded_recovery_bt_node SHARED src/bt.cpp)
target_include_directories(carbot_bounded_recovery_bt_node PRIVATE include)
target_compile_definitions(carbot_bounded_recovery_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(carbot_bounded_recovery_bt_node ${deps})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d layers.xml)
install(TARGETS carbot_recovery_plugins carbot_bounded_recovery_bt_node LIBRARY DESTINATION lib)
install(DIRECTORY behavior_trees config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(recovery_plugin_smoke test/plugin_smoke.cpp)
  ament_target_dependencies(recovery_plugin_smoke ${deps})
  install(TARGETS recovery_plugin_smoke DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
